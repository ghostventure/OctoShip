byte[] Zip(string? special = null, bool missing = false)
{
    using var bytes = new MemoryStream();
    using (var zip = new ZipArchive(bytes, ZipArchiveMode.Create, true))
    {
        if (special != null) { using var writer = new StreamWriter(zip.CreateEntry(special).Open()); writer.Write("test"); }
        else if (!missing)
        {
            zip.CreateEntryFromFile(typeof(MainWindow).Assembly.Location, "OctoShip.dll");
            using var writer = new StreamWriter(zip.CreateEntry("OctoShip.exe").Open()); writer.Write("test apphost placeholder; never executed");
        }
    }
    return bytes.ToArray();
}
