using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Net;
using System.Net.Http;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;
using FileToGitHub;

internal static class Program
{
    private const string Head = "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa";
    private static int checks;
    private static void Check(bool passed, string name) { if (!passed) throw new Exception(name); Console.WriteLine("PASS " + name); checks++; }
    private static async Task Main()
    {
        var unsafePathWarnings = UploadPathAudit.Analyze(new[] { "CON.txt", "space .txt ", "Folder/Name", "folder/name", "invisible\u200Bname", "café.txt", "cafe\u0301.txt", "p\u0430yload.txt" });
        Check(unsafePathWarnings.Any(x => x.Contains("reserved", StringComparison.OrdinalIgnoreCase)), "reserved Windows filename is flagged");
        Check(unsafePathWarnings.Any(x => x.Contains("space or dot", StringComparison.OrdinalIgnoreCase)), "trailing dot or space is flagged");
        Check(unsafePathWarnings.Any(x => x.Contains("letter case", StringComparison.OrdinalIgnoreCase)), "case-only destination collision is flagged");
        Check(unsafePathWarnings.Any(x => x.Contains("invisible", StringComparison.OrdinalIgnoreCase)), "invisible text control is flagged");
        Check(unsafePathWarnings.Any(x => x.Contains("Unicode normalization", StringComparison.OrdinalIgnoreCase)), "non-NFC path is flagged");
        Check(unsafePathWarnings.Any(x => x.Contains("mixes Latin", StringComparison.OrdinalIgnoreCase)), "mixed Latin-Cyrillic filename is flagged");
        Check(UploadPathAudit.Analyze(new[] { "src/Program.cs", "docs/readme.md" }).Count == 0, "ordinary repository paths remain warning-free");
        var dir = Path.Combine(Path.GetTempPath(), "OctoShip-upload-checks-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(dir);
        try
        {
            var local = Path.Combine(dir, "run.sh"); await File.WriteAllTextAsync(local, "echo example");
            var file = new BatchUploadFile(local, "run.sh", "test", new FileInfo(local).Length, File.GetLastWriteTimeUtc(local));
            var handler = new FakeGitHub();
            using var http = new HttpClient(handler);
            var service = new GitHubAtomicBatchService(http);
            var result = await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head,
                validateContent: (_, bytes, _) => { Check(Encoding.UTF8.GetString(bytes) == "echo example", "validator receives exact bytes"); return Task.CompletedTask; });
            Check(result.Count == 1 && result[0].Succeeded, "single batch reports success");
            Check(handler.Calls.Count(x => x.Method == "PATCH") == 1 && handler.Calls.Count(x => x.Path.EndsWith("/git/commits") && x.Method == "POST") == 1, "exactly one commit and ref publication");
            var patch = handler.Calls.Single(x => x.Method == "PATCH");
            Check(!JsonDocument.Parse(patch.Body!).RootElement.GetProperty("force").GetBoolean(), "branch update never forces");
            var tree = JsonDocument.Parse(handler.Calls.Single(x => x.Path.EndsWith("/git/trees") && x.Method == "POST").Body!);
            Check(tree.RootElement.GetProperty("tree")[0].GetProperty("mode").GetString() == "100755", "existing executable mode preserved");
            handler.Reset(); handler.ErrorBody = "{\"message\":\"token=fake-test-credential <script>steal()</script> private-data\"}";
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head); throw new Exception("Expected HTTP failure"); }
            catch (HttpRequestException ex)
            {
                Check(ex.StatusCode == HttpStatusCode.Forbidden && ex.Message.Contains("HTTP 403", StringComparison.Ordinal), "GitHub HTTP failure retains status for actionable diagnostics");
                Check(!ex.Message.Contains("fake-test-credential", StringComparison.Ordinal) && !ex.Message.Contains("<script>", StringComparison.Ordinal) && !ex.Message.Contains("private-data", StringComparison.Ordinal), "GitHub HTTP error body cannot leak credentials, private data, or untrusted HTML into errors");
            }
            handler.Reset(); handler.Moved = true;
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head); throw new Exception("Expected conflict"); }
            catch (InvalidOperationException) { Check(handler.Calls.All(x => x.Method == "GET"), "changed head rejected before writes"); }
            handler.Reset(); handler.Mode = "120000";
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head); throw new Exception("Expected symlink rejection"); }
            catch (InvalidOperationException) { Check(handler.Calls.All(x => x.Method == "GET"), "symlink replacement rejected before writes"); }
            handler.Reset();
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head, validateContent: (_, _, _) => throw new InvalidOperationException("scanner declined")); throw new Exception("Expected validator rejection"); }
            catch (InvalidOperationException) { Check(handler.Calls.All(x => x.Method == "GET"), "content rejection prevents remote writes"); }
            handler.Reset(); handler.MoveAtPublication = true;
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head); throw new Exception("Expected late conflict"); }
            catch (InvalidOperationException) { Check(handler.Calls.All(x => x.Method != "PATCH"), "late branch change prevents publication"); }
            handler.Reset();
            await File.AppendAllTextAsync(local, " changed");
            try { await service.UploadAsync("owner", "repo", "main", "fake-test-credential", new[] { file }, "test", Head); throw new Exception("Expected local conflict"); }
            catch (InvalidOperationException) { Check(handler.Calls.All(x => x.Method == "GET"), "changed local file rejected before writes"); }
            var store = new PersistentUploadQueueStore(Path.Combine(dir, "journal"));
            Directory.CreateDirectory(Path.Combine(dir, "journal"));
            var journalKey = Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes("owner/repo\nmain")));
            var unrelatedTemp = Path.Combine(dir, "journal", journalKey + ".json.tmp");
            await File.WriteAllTextAsync(unrelatedTemp, "keep this unrelated file");
            var row = new PersistentUploadQueueStore.Entry(local, "run.sh", "test", file.Size, file.CapturedLastWriteTimeUtc!.Value, "Uploading", null);
            store.Save(new PersistentUploadQueueStore.Snapshot(1, "owner/repo", "main", 1, "batch test", new[] { row }));
            Check(await File.ReadAllTextAsync(unrelatedTemp) == "keep this unrelated file", "queue save leaves a predictable unrelated temp file untouched");
            Check(!Directory.EnumerateFiles(Path.Combine(dir, "journal"), "*.tmp").Except(new[] { unrelatedTemp }, StringComparer.OrdinalIgnoreCase).Any(), "queue save cleans only its own temporary file");
            var saved = store.Load("owner/repo", "main")!;
            Check(saved.Entries[0].Status == "Uploading" && saved.CommitMode == 1, "queue state and mode survive restart");
            Check(PersistentUploadQueueStore.RestoreItem(saved.Entries[0]).HasChangedOnDisk(), "restored metadata detects changed source");
            Check(store.Load("owner/repo", "other") == null, "queues isolated by branch");
            Check(!File.ReadAllText(Directory.GetFiles(Path.Combine(dir, "journal"), "*.json").Single()).Contains("fake-test-credential"), "journal excludes credential");
            store.Delete("owner/repo", "main"); Check(store.Load("owner/repo", "main") == null, "completed journal can be removed");
            Console.WriteLine($"{checks} upload checks passed. No network requests were sent.");
        }
        finally { Directory.Delete(dir, true); }
    }
    private sealed class FakeGitHub : HttpMessageHandler
    {
        public List<(string Method, string Path, string? Body)> Calls = new();
        public bool Moved; public bool MoveAtPublication; public string Mode = "100755"; public string? ErrorBody; private int heads;
        public void Reset() { Calls.Clear(); Moved = false; MoveAtPublication = false; Mode = "100755"; ErrorBody = null; heads = 0; }
        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            var path = request.RequestUri!.AbsolutePath;
            Calls.Add((request.Method.Method, path, request.Content == null ? null : await request.Content.ReadAsStringAsync()));
            if (ErrorBody != null) return new HttpResponseMessage(HttpStatusCode.Forbidden) { Content = new StringContent(ErrorBody) };
            string json;
            if (path.Contains("/git/ref/heads/")) json = JsonSerializer.Serialize(new { @object = new { sha = Moved || (MoveAtPublication && ++heads > 1) ? "bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb" : Head } });
            else if (request.Method == HttpMethod.Get && path.Contains("/git/commits/")) json = "{\"tree\":{\"sha\":\"base-tree\"}}";
            else if (request.Method == HttpMethod.Get && path.Contains("/git/trees/")) json = JsonSerializer.Serialize(new { truncated = false, tree = new[] { new { path = "run.sh", mode = Mode } } });
            else if (request.Method == HttpMethod.Get) json = "{\"archived\":false,\"disabled\":false,\"permissions\":{\"push\":true}}";
            else json = "{\"sha\":\"cccccccccccccccccccccccccccccccccccccccc\"}";
            return new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent(json) };
        }
    }
}
